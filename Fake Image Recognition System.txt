# Tuned decision threshold for classifying an image as REAL.
# Obtained from tune_threshold.py using validation F1-score.

REAL_IF_HIGH_PROB = True
UNCERTAINTY_MARGIN = 0.08

def calibrate_with_validation_data():
    """Calibrate threshold and label direction from validation data if available."""
    global THRESHOLD, REAL_IF_HIGH_PROB
    if model is None:
        return

    valid_dir = os.path.join(APP_DIR, "..", "dataset", "strict_split", "valid")
    if not os.path.isdir(valid_dir):
        valid_dir = os.path.join(APP_DIR, "..", "dataset", "valid")
    if not os.path.isdir(valid_dir):
        print("Validation folder not found; using default threshold:", THRESHOLD)
        return

    try:
        raw_val_data = tf.keras.utils.image_dataset_from_directory(
            valid_dir,
            labels="inferred",
            label_mode="int",
            image_size=TARGET_SIZE,
            batch_size=32,
            shuffle=False,
        )

        val_data = raw_val_data.map(
            lambda x, y: (preprocess_input(tf.cast(x, tf.float32)), tf.cast(y, tf.float32)),
            num_parallel_calls=tf.data.AUTOTUNE,
        ).prefetch(tf.data.AUTOTUNE)

        probs = model.predict(val_data, verbose=0).ravel()
        y_true = np.concatenate([y.numpy() for _, y in raw_val_data], axis=0).astype(int)

        thresholds = np.arange(0.1, 0.91, 0.01)
        best_acc = -1.0
        best_t = THRESHOLD
        best_real_if_high = True

        for t in thresholds:
            pred_normal = (probs > t).astype(int)
            acc_normal = float((pred_normal == y_true).mean())

            pred_inverted = 1 - pred_normal
            acc_inverted = float((pred_inverted == y_true).mean())

            if acc_normal > best_acc:
                best_acc = acc_normal
                best_t = float(t)
                best_real_if_high = True

            if acc_inverted > best_acc:
                best_acc = acc_inverted
                best_t = float(t)
                best_real_if_high = False

        THRESHOLD = best_t
        REAL_IF_HIGH_PROB = best_real_if_high
        print(
            "Calibration complete. threshold=",
            round(THRESHOLD, 3),
            "real_if_high_prob=",
            REAL_IF_HIGH_PROB,
            "val_acc=",
            round(best_acc * 100.0, 2),
            "%",
        )
    except Exception as e:
        print(f"Calibration skipped due to error: {e}")


if ENABLE_RUNTIME_CALIBRATION:
    calibrate_with_validation_data()
else:
    print("Runtime threshold calibration disabled. Using fixed threshold:", THRESHOLD)


def predict_image(img_path):
    """Return predicted label, confidence, and diagnostics for UI display."""
    if model is None:
        raise RuntimeError(
            "Model is not loaded. Place the model file in the 'models' folder."
        )

    img = load_img(img_path, target_size=TARGET_SIZE)
    base = img_to_array(img)

    # Keep inference consistent with eval_valid.py and tune_threshold.py: single-view prediction.
    batch = np.expand_dims(base, axis=0).astype(np.float32)
    batch = preprocess_input(batch)

    decision_prob = float(model.predict(batch, verbose=0).ravel()[0])
    prob_mean = decision_prob
    prob_median = decision_prob
    prob_std = 0.0
    print("Prediction prob:", decision_prob)
    predicts_real = decision_prob > THRESHOLD if REAL_IF_HIGH_PROB else decision_prob <= THRESHOLD

    prob_real = decision_prob if REAL_IF_HIGH_PROB else (1.0 - decision_prob)
    prob_fake = 1.0 - prob_real

    if predicts_real:
        label = "Real"
        confidence = prob_real * 100.0
    else:
        label = "Fake"
        confidence = prob_fake * 100.0

    note = f"Model prob: {decision_prob:.3f}"
    return label, round(confidence, 2), note

import argparse
import csv
import hashlib
import json
import os
import random
import shutil
from datetime import datetime

SUPPORTED_EXTS = (".bmp", ".gif", ".jpeg", ".jpg", ".png", ".webp", ".tif", ".tiff")
CLASS_NAMES = ("fake", "real")


def parse_args():
    app_dir = os.path.dirname(os.path.abspath(__file__))
    project_root = os.path.dirname(app_dir)
    dataset_root = os.path.join(project_root, "dataset")

    parser = argparse.ArgumentParser(
        description=(

        )
    )
    parser.add_argument(

        default=dataset_root,
        help="Dataset root containing source folders.",

    parser.add_argument(
        "--source-mode",
        choices=["existing-splits", "raw-classes"],
        default="existing-splits",
        help=(
            "existing-splits: source from dataset/train|valid|test/{fake,real}; "
            "raw-classes: source from dataset/{fake,real}."
        ),
    )
    parser.add_argument(
        "--output-dir",
        default=None,
        help="Output split root. Default: <dataset-root>/strict_split",
    )
    parser.add_argument("--seed", type=int, default=42, help="Random seed for reproducible split.")
    parser.add_argument("--valid-ratio", type=float, default=0.15, help="Validation ratio in [0,1).")
    parser.add_argument("--test-ratio", type=float, default=0.10, help="Test ratio in [0,1).")
    parser.add_argument(
        "--valid-count-per-class",
        type=int,
        default=None,
        help="If set, use this exact validation count per class (overrides --valid-ratio).",
    )
    parser.add_argument(
        "--test-count-per-class",

        default=None,
        help="If set, use this exact test count per class (overrides --test-ratio).",

    parser.add_argument(
        "--link-mode",

        action="store_true",
        help="Delete output directory first if it already exists.",
    )
    return parser.parse_args()


def collect_sources(dataset_root, source_mode):
    sources = {"fake": [], "real": []}

    if source_mode == "raw-classes":
        for cls in CLASS_NAMES:
            class_dir = os.path.join(dataset_root, cls)
            if not os.path.isdir(class_dir):
                raise FileNotFoundError(f"Expected source class folder: {class_dir}")
            sources[cls].append(class_dir)
    else:
        found_any = False
        for split in ("train", "valid", "test"):
            split_dir = os.path.join(dataset_root, split)
            if not os.path.isdir(split_dir):
                continue
            for cls in CLASS_NAMES:
                class_dir = os.path.join(split_dir, cls)
                if os.path.isdir(class_dir):
                    found_any = True
                    sources[cls].append(class_dir)
        if not found_any:
            raise FileNotFoundError(
                "No existing split source directories found under dataset/{train,valid,test}/{fake,real}."
            )

    return sources


def collect_images(class_dirs):
    paths = []
    for class_dir in class_dirs:
        for root, _, files in os.walk(class_dir):
            for name in files:
                ext = os.path.splitext(name)[1].lower()
                if ext in SUPPORTED_EXTS:
                    paths.append(os.path.abspath(os.path.join(root, name)))

    deduped = sorted(set(paths))
    return deduped


def stratified_split(
    paths_by_class,
    valid_ratio,
    test_ratio,
    seed,
    valid_count_per_class=None,
    test_count_per_class=None,
):
    if valid_ratio < 0 or test_ratio < 0 or (valid_ratio + test_ratio) >= 1.0:
        raise ValueError("valid_ratio and test_ratio must be >=0 and sum to < 1.0")
    if valid_count_per_class is not None and valid_count_per_class < 0:
        raise ValueError("valid_count_per_class must be >= 0")
    if test_count_per_class is not None and test_count_per_class < 0:
        raise ValueError("test_count_per_class must be >= 0")

    rng = random.Random(seed)
    split_map = {"train": {"fake": [], "real": []}, "valid": {"fake": [], "real": []}, "test": {"fake": [], "real": []}}

    for cls, paths in paths_by_class.items():
        local = list(paths)
        rng.shuffle(local)
        n = len(local)

        if test_count_per_class is None:
            n_test = int(round(n * test_ratio))
        else:
            n_test = int(test_count_per_class)

        if valid_count_per_class is None:
            n_valid = int(round(n * valid_ratio))
        else:
            n_valid = int(valid_count_per_class)

        if n_test + n_valid >= n:
            raise RuntimeError(
                f"Requested split sizes exceed available samples for class '{cls}': "
                f"n={n}, valid={n_valid}, test={n_test}."
            )

        n_train = n - n_valid - n_test

        if n_train <= 0:
            raise RuntimeError(
                f"Not enough samples for class '{cls}' with ratios valid={valid_ratio}, test={test_ratio}."
            )

        split_map["test"][cls] = local[:n_test]
        split_map["valid"][cls] = local[n_test:n_test + n_valid]
        split_map["train"][cls] = local[n_test + n_valid:]

    return split_map


def ensure_clean_dir(path, overwrite):
    if os.path.exists(path):
        if not overwrite:
            raise FileExistsError(
                f"Output directory already exists: {path}. Use --overwrite to replace it."
            )
        shutil.rmtree(path)
    os.makedirs(path, exist_ok=True)


def safe_name(src_path, index):
    base = os.path.basename(src_path)
    stem, ext = os.path.splitext(base)
    digest = hashlib.sha1(src_path.encode("utf-8")).hexdigest()[:10]
    return f"{index:06d}_{stem}_{digest}{ext.lower()}"


def materialize(split_map, output_root, link_mode):
    manifests_dir = os.path.join(output_root, "manifests")
    os.makedirs(manifests_dir, exist_ok=True)

    summary = {"train": {}, "valid": {}, "test": {}}

    for split in ("train", "valid", "test"):
        rows = []
        for cls in CLASS_NAMES:
            dst_class_dir = os.path.join(output_root, split, cls)
            os.makedirs(dst_class_dir, exist_ok=True)

            class_paths = split_map[split][cls]
            for idx, src in enumerate(class_paths, start=1):
                dst_name = safe_name(src, idx)
                dst = os.path.join(dst_class_dir, dst_name)
                if link_mode == "hardlink":
                    try:
                        os.link(src, dst)
                    except OSError:
                        shutil.copy2(src, dst)
                else:
                    shutil.copy2(src, dst)

                rows.append(
                    {
                        "split": split,
                        "label": 0 if cls == "fake" else 1,
                        "label_str": cls,
                        "src_path": src,
                        "dst_path": dst,
                    }
                )

            summary[split][cls] = len(class_paths)

        manifest_path = os.path.join(manifests_dir, f"{split}.csv")
        with open(manifest_path, "w", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(
                f,
                fieldnames=["split", "label", "label_str", "src_path", "dst_path"],
            )
            writer.writeheader()
            writer.writerows(rows)

    return summary


def main():
    args = parse_args()
    dataset_root = os.path.abspath(args.dataset_root)
    output_root = os.path.abspath(args.output_dir or os.path.join(dataset_root, "strict_split"))

    sources = collect_sources(dataset_root, args.source_mode)
    paths_by_class = {
        cls: collect_images(sources[cls])
        for cls in CLASS_NAMES
    }

    for cls in CLASS_NAMES:
        if len(paths_by_class[cls]) == 0:
            raise RuntimeError(f"No usable images found for class '{cls}'.")

    split_map = stratified_split(
        paths_by_class=paths_by_class,
        valid_ratio=float(args.valid_ratio),
        test_ratio=float(args.test_ratio),
        seed=int(args.seed),
        valid_count_per_class=args.valid_count_per_class,
        test_count_per_class=args.test_count_per_class,
    )

    ensure_clean_dir(output_root, overwrite=bool(args.overwrite))
    summary = materialize(split_map, output_root=output_root, link_mode=args.link_mode)

    metadata = {
        "created_at": datetime.utcnow().isoformat() + "Z",
        "dataset_root": dataset_root,
        "output_root": output_root,
        "source_mode": args.source_mode,
        "seed": int(args.seed),
        "valid_ratio": float(args.valid_ratio),
        "test_ratio": float(args.test_ratio),
        "valid_count_per_class": args.valid_count_per_class,
        "test_count_per_class": args.test_count_per_class,
        "link_mode": args.link_mode,
        "source_counts": {cls: len(paths_by_class[cls]) for cls in CLASS_NAMES},
        "split_counts": summary,
    }

    metadata_path = os.path.join(output_root, "manifests", "metadata.json")
    with open(metadata_path, "w", encoding="utf-8") as f:
        json.dump(metadata, f, indent=2)

    print("Created strict stratified split at:", output_root)
    print("Source counts:", metadata["source_counts"])
    print("Split counts:", summary)
    print("Metadata:", metadata_path)


if __name__ == "__main__":
    main()

def collect_candidate_files(class_dir):
    paths = []
    unsupported = []
    for root, _, files in os.walk(class_dir):
        for name in files:
            ext = os.path.splitext(name)[1].lower()
            if ext in SUPPORTED_EXTS:
                paths.append(os.path.join(root, name))
            else:
                unsupported.append(os.path.join(root, name))
    paths.sort()
    unsupported.sort()
    return paths, unsupported

    if unsupported_total > 0:

            "Warning: Found",
            unsupported_total,
            "unsupported files in class folders (ignored by TensorFlow).",
        )

    all_paths = fake_candidates + real_candidates
    all_labels = [0] * len(fake_candidates) + [1] * len(real_candidates)

    good_probs = []

    batch_labels = []
    start_time = time.perf_counter()
    processed_files = 0

    def _fmt_duration(seconds):
        seconds = max(0, int(seconds))
        hrs = seconds // 3600
        mins = (seconds % 3600) // 60
        secs = seconds % 60
        if hrs > 0:
            return f"{hrs:02d}:{mins:02d}:{secs:02d}"
        return f"{mins:02d}:{secs:02d}"

    def _print_progress(force=False):
        if total_files == 0:
            return
        if (not force) and (processed_files % progress_every != 0):
            return
        elapsed = time.perf_counter() - start_time
        rate = processed_files / elapsed if elapsed > 0 else 0.0
        remaining = total_files - processed_files
        eta = (remaining / rate) if rate > 0 else 0.0
        pct = (processed_files / total_files) * 100.0
        print(
            f"Progress [{split_name}]: {processed_files}/{total_files} "
            f"({pct:.1f}%) | loaded={len(good_true)} bad={len(bad_paths)} "
            f"| {rate:.2f} img/s | elapsed={_fmt_duration(elapsed)} "
            f"ETA={_fmt_duration(eta)}"
        )

    def flush_batch():
        if not batch_imgs:
            return
        x = np.stack(batch_imgs, axis=0).astype(np.float32)
        x = preprocess_input(x)
        probs_batch = model.predict(x, verbose=0).ravel()
        good_probs.extend(probs_batch.tolist())
        good_true.extend(batch_labels)
        batch_imgs.clear()
        batch_labels.clear()

    for path, label in zip(all_paths, all_labels):
        try:
            img = tf.keras.utils.load_img(path, target_size=img_size)
            arr = tf.keras.utils.img_to_array(img)
            batch_imgs.append(arr)
            batch_labels.append(label)
            if len(batch_imgs) >= batch_size:
                flush_batch()
        except Exception:
            bad_paths.append(path)
        finally:
            processed_files += 1
            _print_progress(force=(processed_files == total_files))

    flush_batch()
    _print_progress(force=True)

    if bad_paths:
        os.makedirs(os.path.dirname(bad_files_report), exist_ok=True)
        with open(bad_files_report, "a", encoding="utf-8") as f:
            f.write(f"\n[{split_name}] unreadable images in {eval_dir}\n")
            for p in bad_paths:
                f.write(p + "\n")

    loaded_count = len(good_true)
    print("Decodable images loaded:", loaded_count)
    if bad_paths:
        print(
            "Warning: Found",
            len(bad_paths),
            "unreadable image files. Paths appended to:",
            bad_files_report,
        )

    if loaded_count != candidate_count:
        msg = (
            f"Decodable image count mismatch for '{split_name}'. "
            f"Candidate image files: {candidate_count}, decodable: {loaded_count}."
        )
        if allow_mismatch:
            print("Warning:", msg)
        else:
            raise RuntimeError(msg)

    if loaded_count == 0:
        raise RuntimeError(f"No decodable images found in '{eval_dir}'.")

    true = np.array(good_true, dtype=np.int32)
    probs = np.array(good_probs, dtype=np.float32)
    preds = (probs > threshold).astype(np.int32)

    bce = tf.keras.losses.binary_crossentropy(
        tf.convert_to_tensor(true, dtype=tf.float32),
        tf.convert_to_tensor(probs, dtype=tf.float32),
    )
    loss = float(tf.reduce_mean(bce).numpy())
    acc = float((preds == true).mean())

    print("Loss:", loss)
    print("Accuracy:", acc * 100.0, "%")

    cm = np.zeros((2, 2), dtype=int)
    for t, p in zip(true, preds):
        row = 0 if int(t) == 0 else 1
        col = 0 if int(p) == 0 else 1
        cm[row, col] += 1

    tn, fp, fn, tp = cm[0, 0], cm[0, 1], cm[1, 0], cm[1, 1]
    print("\nConfusion Matrix (counts):")
    print("            Pred Fake   Pred Real")
    print(f"True Fake   {tn:9d} {fp:10d}")
    print(f"True Real   {fn:9d} {tp:10d}")

    acc_manual = (preds == true).mean() * 100.0
    print("Manual accuracy check:", acc_manual, "%")

    return {
        "split": split_name,
        "loss": float(loss),
        "accuracy": float(acc),
        "loaded_count": loaded_count,
        "expected_supported": candidate_count,
        "bad_count": len(bad_paths),
    }

def run_step(cmd, cwd):
    print("\n>>>", " ".join(cmd))

    args = parse_args()

import argparse
import csv
import os
import shutil
import time
from typing import List, Tuple
import numpy as np
import tensorflow as tf
from sklearn.metrics import f1_score, recall_score

ImageDataGenerator = tf.keras.preprocessing.image.ImageDataGenerator
preprocess_input = tf.keras.applications.efficientnet.preprocess_input

APP_DIR = os.path.dirname(os.path.abspath(__file__))
PROJECT_ROOT = os.path.dirname(APP_DIR)
MODEL_PATH = os.path.join(APP_DIR, "models", "best_model.keras")
DATASET_ROOT = os.path.join(PROJECT_ROOT, "dataset")
DEFAULT_SOURCE = os.path.join(DATASET_ROOT, "valid")
HARD_REAL_DIR = os.path.join(DATASET_ROOT, "hard_examples", "real")
REPORT_PATH = os.path.join(APP_DIR, "hard_examples_report.csv")


def parse_args():
    parser = argparse.ArgumentParser(description="Mine false-positive real images for re-training.")
    parser.add_argument("--source", default=DEFAULT_SOURCE, help="Directory containing real/fake subfolders to scan.")
    parser.add_argument("--threshold", type=float, default=None, help="Decision threshold. If omitted, macro F1 sweep determines it.")
    parser.add_argument("--top-k", type=int, default=400, help="Maximum number of real hard examples to keep.")
    parser.add_argument("--min-prob", type=float, default=0.55, help="Minimum fake probability required to keep a real sample.")
    parser.add_argument("--batch-size", type=int, default=32, help="Batch size for inference.")
    return parser.parse_args()


def _macro_f1(y_true, y_pred):
    f1_fake = float(f1_score(y_true, y_pred, pos_label=0, zero_division=0))
    f1_real = float(f1_score(y_true, y_pred, pos_label=1, zero_division=0))
    real_recall = float(recall_score(y_true, y_pred, pos_label=1, zero_division=0))
    return (f1_real + f1_fake) / 2.0, f1_real, f1_fake, real_recall


def _sweep_threshold(y_true, probs):
    best = {
        "threshold": 0.5,
        "macro_f1": 0.0,
        "real_f1": 0.0,
        "fake_f1": 0.0,
        "real_recall": 0.0,
    }
    for t in np.arange(0.1, 0.9, 0.01):
        y_pred = (probs > t).astype(int)
        macro_f1, real_f1, fake_f1, real_recall = _macro_f1(y_true, y_pred)
        if macro_f1 > best["macro_f1"]:
            best.update({
                "threshold": float(t),
                "macro_f1": macro_f1,
                "real_f1": real_f1,
                "fake_f1": fake_f1,
                "real_recall": real_recall,
            })
    return best


def _ensure_model():
    if not os.path.exists(MODEL_PATH):
        raise FileNotFoundError(f"Model not found at {MODEL_PATH}. Train first.")


def _copy_hard_examples(records: List[Tuple[str, float]]):
    os.makedirs(HARD_REAL_DIR, exist_ok=True)
    copied = []
    timestamp = int(time.time())
    for idx, (src, prob) in enumerate(records):
        ext = os.path.splitext(src)[1].lower() or ".jpg"
        dst_name = f"mined_real_{timestamp}_{idx:05d}_p{prob:.3f}{ext}"
        dst_path = os.path.join(HARD_REAL_DIR, dst_name)
        shutil.copy2(src, dst_path)
        copied.append((src, dst_path, prob))
    return copied


def _write_report(rows):
    headers = ["source_path", "copied_path", "predicted_prob"]
    with open(REPORT_PATH, "w", newline="", encoding="utf-8") as handle:
        writer = csv.writer(handle)
        writer.writerow(headers)
        writer.writerows(rows)


def main():
    args = parse_args()
    _ensure_model()
    if not os.path.isdir(args.source):
        raise FileNotFoundError(f"Source directory not found: {args.source}")

    print("Loading model from:", MODEL_PATH)
    model = tf.keras.models.load_model(MODEL_PATH)
    img_size = model.input_shape[1] if hasattr(model, "input_shape") else 300

    datagen = ImageDataGenerator(preprocessing_function=preprocess_input)
    data = datagen.flow_from_directory(
        args.source,
        target_size=(img_size, img_size),
        batch_size=args.batch_size,
        class_mode='binary',
        shuffle=False,
    )

    print("Running inference over", len(data.filenames), "images...")
    probs = model.predict(data).ravel()
    y_true = data.classes

    if args.threshold is None:
        sweep = _sweep_threshold(y_true, probs)
        threshold = sweep["threshold"]
        print(
            "Sweep best -> threshold:", round(sweep["threshold"], 3),
            "macro F1:", round(sweep["macro_f1"], 4),
            "real F1:", round(sweep["real_f1"], 4),
            "fake F1:", round(sweep["fake_f1"], 4),
            "real recall:", round(sweep["real_recall"], 4),
        )
    else:
        threshold = args.threshold
        print("Using provided threshold:", threshold)

    filepaths = data.filepaths
    hard_real = []
    for path, label, prob in zip(filepaths, y_true, probs):
        if label == 1 and prob <= threshold and (1.0 - prob) >= args.min_prob:
            hard_real.append((path, float(prob)))

    hard_real.sort(key=lambda item: item[1], reverse=True)
    if args.top_k > 0:
        hard_real = hard_real[:args.top_k]

    if not hard_real:
        print("No hard real samples met the criteria.")
        return

    copied = _copy_hard_examples(hard_real)
    rows = [(src, dst, f"{prob:.4f}") for src, dst, prob in copied]
    _write_report(rows)

    print("Copied", len(copied), "hard real samples to", HARD_REAL_DIR)
    print("Report written to:", REPORT_PATH)


if __name__ == "__main__":
    main()

import argparse
import os
import numpy as np
import tensorflow as tf

ImageDataGenerator = tf.keras.preprocessing.image.ImageDataGenerator
preprocess_input = tf.keras.applications.efficientnet.preprocess_input

APP_DIR = os.path.dirname(os.path.abspath(__file__))
PROJECT_ROOT = os.path.dirname(APP_DIR)
MODEL_PATH = os.path.join(APP_DIR, "models", "best_model.keras")
VALID_DIR = os.path.join(PROJECT_ROOT, "dataset", "valid")


def _metrics(y_true, y_pred):
    tn = int(((y_true == 0) & (y_pred == 0)).sum())
    fp = int(((y_true == 0) & (y_pred == 1)).sum())
    fn = int(((y_true == 1) & (y_pred == 0)).sum())
    tp = int(((y_true == 1) & (y_pred == 1)).sum())

    acc = float((y_pred == y_true).mean())

    precision_fake = tn / (tn + fn) if (tn + fn) else 0.0
    recall_fake = tn / (tn + fp) if (tn + fp) else 0.0
    f1_fake = (
        2 * precision_fake * recall_fake / (precision_fake + recall_fake)
        if (precision_fake + recall_fake)
        else 0.0
    )

    precision_real = tp / (tp + fp) if (tp + fp) else 0.0
    recall_real = tp / (tp + fn) if (tp + fn) else 0.0
    f1_real = (
        2 * precision_real * recall_real / (precision_real + recall_real)
        if (precision_real + recall_real)
        else 0.0
    )

    macro_f1 = (f1_fake + f1_real) / 2.0

    return {
        "tn": tn,
        "fp": fp,
        "fn": fn,
        "tp": tp,
        "acc": acc,
        "precision_fake": precision_fake,
        "recall_fake": recall_fake,
        "f1_fake": f1_fake,
        "precision_real": precision_real,
        "recall_real": recall_real,
        "f1_real": f1_real,
        "macro_f1": macro_f1,
    }


def _print_block(title, threshold, m):
    print("\n" + title)
    print("threshold:", round(float(threshold), 3))
    print("accuracy:", round(m["acc"] * 100.0, 2), "%")
    print("macro_f1:", round(m["macro_f1"], 4))
    print("f1_fake:", round(m["f1_fake"], 4), "| f1_real:", round(m["f1_real"], 4))
    print("\nConfusion Matrix (counts):")
    print("            Pred Fake   Pred Real")
    print(f"True Fake   {m['tn']:9d} {m['fp']:10d}")
    print(f"True Real   {m['fn']:9d} {m['tp']:10d}")


def parse_args():
    parser = argparse.ArgumentParser(
        description="Sanity checks for class mapping, counts, and threshold tuning."
    )
    parser.add_argument(
        "--objective",
        choices=["macro-f1", "real-safe"],
        default="macro-f1",
        help="Optimization objective for tuned threshold.",
    )
    parser.add_argument(
        "--min-real-precision",
        type=float,
        default=0.75,
        help="Used for real-safe objective. Keeps thresholds with precision_real >= this value and chooses best recall_real.",
    )
    parser.add_argument("--batch-size", type=int, default=32)
    return parser.parse_args()


def main():
    args = parse_args()

    if not os.path.exists(MODEL_PATH):
        raise FileNotFoundError(f"Model not found at {MODEL_PATH}.")
    if not os.path.isdir(VALID_DIR):
        raise FileNotFoundError(f"Validation directory not found at {VALID_DIR}.")

    print("Loading model from:", MODEL_PATH)
    model = tf.keras.models.load_model(MODEL_PATH)

    if not hasattr(model, "input_shape") or len(model.input_shape) != 4:
        raise RuntimeError("Unexpected model input shape; expected rank-4 image input.")

    img_size = (int(model.input_shape[1]), int(model.input_shape[2]))
    print("Using validation image size:", img_size)

    val_data = ImageDataGenerator(preprocessing_function=preprocess_input).flow_from_directory(
        VALID_DIR,
        target_size=img_size,
        batch_size=args.batch_size,
        class_mode="binary",
        classes=["fake", "real"],
        shuffle=False,
    )

    expected = {"fake": 0, "real": 1}
    if val_data.class_indices != expected:
        raise RuntimeError(
            f"Unexpected class mapping: {val_data.class_indices}. Expected: {expected}"
        )

    y_true = val_data.classes.astype(int)
    counts = np.bincount(y_true)

    print("\nClass indices:", val_data.class_indices)
    print("Sample counts -> fake:", int(counts[0]) if len(counts) > 0 else 0, ", real:", int(counts[1]) if len(counts) > 1 else 0)

    probs = model.predict(val_data, verbose=0).ravel()

    baseline_pred = (probs > 0.5).astype(int)
    baseline = _metrics(y_true, baseline_pred)
    _print_block("Baseline @ 0.5", 0.5, baseline)

    thresholds = np.arange(0.1, 0.901, 0.01)

    best_t = 0.5
    best_m = baseline

    if args.objective == "macro-f1":
        for t in thresholds:
            y_pred = (probs > t).astype(int)
            m = _metrics(y_true, y_pred)
            if m["macro_f1"] > best_m["macro_f1"]:
                best_t = float(t)
                best_m = m
    else:
        found = False
        for t in thresholds:
            y_pred = (probs > t).astype(int)
            m = _metrics(y_true, y_pred)
            if m["precision_real"] >= args.min_real_precision:
                if (not found) or (m["recall_real"] > best_m["recall_real"]):
                    found = True
                    best_t = float(t)
                    best_m = m
        if not found:
            print(
                "\nNo threshold satisfied precision_real >=",
                args.min_real_precision,
                "; falling back to macro-f1 objective.",
            )
            best_t = 0.5
            best_m = baseline
            for t in thresholds:
                y_pred = (probs > t).astype(int)
                m = _metrics(y_true, y_pred)
                if m["macro_f1"] > best_m["macro_f1"]:
                    best_t = float(t)
                    best_m = m

    _print_block("Tuned Threshold", best_t, best_m)

    print("\nRecommended threshold for app.py:", round(best_t, 3))
    print("Expected direction: REAL_IF_HIGH_PROB = True")


if __name__ == "__main__":
    main()

    else:
        target_size = (300, 300)

    recall = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0

## Reproducible Strict Pipeline

This project supports a reproducible stratified train/valid/test workflow.

### 1) Build deterministic split

```bash
python app/build_stratified_splits.py \
	--dataset-root dataset \
	--source-mode existing-splits \
	--output-dir dataset/strict_split \
	--seed 42 \
	--valid-ratio 0.15 \
	--test-ratio 0.10 \
	--link-mode hardlink \
	--overwrite
```

This creates:

- dataset/strict_split/train/fake, dataset/strict_split/train/real
- dataset/strict_split/valid/fake, dataset/strict_split/valid/real
- dataset/strict_split/test/fake, dataset/strict_split/test/real
- dataset/strict_split/manifests/{train.csv,valid.csv,test.csv,metadata.json}

### 2) Train on strict split

```bash
python app/train.py \
	--dataset-root dataset/strict_split \
	--model-path app/models/best_model.keras \
	--seed 42 \
	--img-size 300 \
	--batch-size 16 \
	--epochs-phase1 10 \
	--epochs-phase2 6
```

Training writes run metadata to app/models/last_train_run.json.

### 3) Tune threshold on validation only

```bash
python app/tune_threshold.py \
	--dataset-root dataset/strict_split \
	--model-path app/models/best_model.keras
```

### 4) Strict test evaluation

```bash
python app/eval_valid.py \
	--dataset-root dataset/strict_split \
	--split test \
	--model-path app/models/best_model.keras \
	--progress-every 100
```

### One-command pipeline

```bash
python app/run_strict_pipeline.py --overwrite-split
```
